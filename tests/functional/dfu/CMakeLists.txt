# ======== Test images ========
#
# Firmware images signed at configure time with MCUboot's imgtool and the committed development key,
# from a small payload each, and embedded as arrays in a generated header (dfu_test_images.h). They
# differ in the identity TLV they carry: this unit's, another product's, another board's, a malformed
# one, none at all, and one whose protected TLVs are more than a unit accepts.

set(TEST_PRODUCT_ID 4660)
set(TEST_PAYLOAD_SIZE 6000)
# What the slots of boards/native_sim.overlay are; imgtool wants the size to know what fits.
set(TEST_SLOT_SIZE 0x10000)

if(DEFINED ZEPHYR_MCUBOOT_MODULE_DIR AND EXISTS "${ZEPHYR_MCUBOOT_MODULE_DIR}/scripts/imgtool.py")
    set(IMGTOOL ${PYTHON_EXECUTABLE} ${ZEPHYR_MCUBOOT_MODULE_DIR}/scripts/imgtool.py)
else()
    find_program(IMGTOOL_PROGRAM imgtool)
    if(NOT IMGTOOL_PROGRAM)
        message(FATAL_ERROR "imgtool is needed to sign the test images: the mcuboot module's scripts/imgtool.py, or imgtool on PATH.")
    endif()
    set(IMGTOOL ${IMGTOOL_PROGRAM})
endif()

set(SIGNING_KEY ${RT_CORE_DIR}/keys/dev-ec-p256.pem)
set(IMAGE_DIR ${CMAKE_CURRENT_BINARY_DIR}/test_images)
file(MAKE_DIRECTORY ${IMAGE_DIR})
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SIGNING_KEY})

set(IMAGES_HEADER ${IMAGE_DIR}/dfu_test_images.h)
file(WRITE ${IMAGES_HEADER}
"// Generated by CMakeLists.txt: the signed test images.
#pragma once

#include <cstdint>

namespace dfu_test::images {

")

# sign_test_image(<name> <payload seed> <version> [imgtool arguments...])
#
# Signs a payload of TEST_PAYLOAD_SIZE bytes built from <payload seed> and appends to the header:
# <NAME> (the image file), <NAME>_SHA256 (the hash of the file as hex text, which the upload sends as
# `sha`) and <NAME>_VERSION (the version img_mgmt lists).
function(sign_test_image name seed version)
    set(payload ${IMAGE_DIR}/${name}.payload.bin)
    set(image ${IMAGE_DIR}/${name}.bin)

    string(LENGTH "${seed}" seed_length)
    math(EXPR repeats "${TEST_PAYLOAD_SIZE} / ${seed_length} + 1")
    string(REPEAT "${seed}" ${repeats} pattern)
    string(SUBSTRING "${pattern}" 0 ${TEST_PAYLOAD_SIZE} pattern)
    file(WRITE ${payload} "${pattern}")

    execute_process(
        COMMAND ${IMGTOOL} sign
            --key ${SIGNING_KEY}
            --header-size 0x200
            --pad-header
            --align 4
            --version ${version}
            --slot-size ${TEST_SLOT_SIZE}
            ${ARGN}
            ${payload} ${image}
        RESULT_VARIABLE result
        OUTPUT_VARIABLE output
        ERROR_VARIABLE output
    )
    if(NOT result EQUAL 0)
        message(FATAL_ERROR "imgtool failed to sign the test image ${name}:\n${output}")
    endif()

    file(READ ${image} hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
    file(SHA256 ${image} sha256)

    # The version as img_mgmt prints it: major.minor.revision, and .build when it is not 0.
    string(REPLACE "+" "." printed_version "${version}")
    string(REGEX REPLACE "\\.0$" "" printed_version "${printed_version}")

    string(TOUPPER ${name} upper)
    file(APPEND ${IMAGES_HEADER}
"inline constexpr uint8_t ${upper}[] = {${bytes}};
inline constexpr char ${upper}_SHA256[] = \"${sha256}\";
inline constexpr char ${upper}_VERSION[] = \"${printed_version}\";

")
endfunction()

eerie_leap_image_identity_hex(${TEST_PRODUCT_ID} "${CONFIG_BOARD_TARGET}" own_identity)

math(EXPR other_product "${TEST_PRODUCT_ID} + 1")
eerie_leap_image_identity_hex(${other_product} "${CONFIG_BOARD_TARGET}" other_product_identity)
eerie_leap_image_identity_hex(${TEST_PRODUCT_ID} "other_board/soc/core" other_board_identity)

# A TLV that is not an identity: a wrong format byte, and a board length that exceeds the value.
set(malformed_identity "0x02a8")
# 130 bytes of another protected TLV, so that the protected area is above the 128 bytes a unit accepts.
string(REPEAT "ab" 130 filler_hex)

sign_test_image(good "0123456789abcdef" 1.2.3+4 --custom-tlv ${EERIE_LEAP_IMAGE_IDENTITY_TLV_TAG} ${own_identity})
sign_test_image(good_other "fedcba9876543210" 1.2.4+5 --custom-tlv ${EERIE_LEAP_IMAGE_IDENTITY_TLV_TAG} ${own_identity})
sign_test_image(wrong_product "0123456789abcdef" 1.2.3+4 --custom-tlv ${EERIE_LEAP_IMAGE_IDENTITY_TLV_TAG} ${other_product_identity})
sign_test_image(wrong_board "0123456789abcdef" 1.2.3+4 --custom-tlv ${EERIE_LEAP_IMAGE_IDENTITY_TLV_TAG} ${other_board_identity})
sign_test_image(bad_identity "0123456789abcdef" 1.2.3+4 --custom-tlv ${EERIE_LEAP_IMAGE_IDENTITY_TLV_TAG} ${malformed_identity})
sign_test_image(no_identity "0123456789abcdef" 1.2.3+4)
sign_test_image(
    oversize_tlvs "0123456789abcdef" 1.2.3+4
    --custom-tlv ${EERIE_LEAP_IMAGE_IDENTITY_TLV_TAG} ${own_identity}
    --custom-tlv 0xa1 0x${filler_hex}
)

    DFU_TEST_PRODUCT_ID=${TEST_PRODUCT_ID}
